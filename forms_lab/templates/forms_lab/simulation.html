How Django Validates This Form

Step 1: User Enters Data

The user enters information into the Django form.

Step 2: Form is Submitted

The form sends the entered data to the Django server using a POST request. 

POST request

Step 3: Django Creates the Form

Django receives the submitted data and creates a StudentForm object. 

form = StudentForm(request.POST)

Step 4: Django Validates the Form

Django checks each field according to the validation rules defined in forms.py. 

form.is_valid()

Django checks things such as required fields, email format, minimum values, maximum values, minimum length and maximum length.

Step 5: Validation Result

If the data is invalid, Django generates validation errors.

If the data is valid, Django stores the validated information in cleaned_data. 

form.errors

form.cleaned_data

Django Code

from django import forms


class StudentForm(forms.Form):

    name = forms.CharField(
        label="Full Name",
        max_length=50,
        min_length=3,
        required=True
    )

    email = forms.EmailField(
        label="Email Address",
        required=True
    )

    age = forms.IntegerField(
        label="Age",
        min_value=18,
        max_value=60,
        required=True
    )

    phone = forms.CharField(
        label="Phone Number",
        max_length=10,
        min_length=10,
        required=True
    )

    gender = forms.ChoiceField(
        label="Gender",
        choices=[
            ("Male", "Male"),
            ("Female", "Female"),
            ("Other", "Other")
        ],
        required=True
    )

    department = forms.ChoiceField(
        label="Department",
        choices=[
            ("Computer", "Computer Engineering"),
            ("IT", "Information Technology"),
            ("EXTC", "Electronics and Telecommunication"),
            ("AI", "Artificial Intelligence")
        ],
        required=True
    )

    year = forms.ChoiceField(
        label="Year",
        choices=[
            ("1", "First Year"),
            ("2", "Second Year"),
            ("3", "Third Year"),
            ("4", "Fourth Year")
        ],
        required=True
    )

    cgpa = forms.FloatField(
        label="CGPA",
        min_value=0,
        max_value=10,
        required=True
    )

    website = forms.URLField(
        label="Portfolio Website",
        required=False
    )

    date_of_birth = forms.DateField(
        label="Date of Birth",
        required=True
    )

    address = forms.CharField(
        label="Address",
        max_length=200,
        required=True
    )

    password = forms.CharField(
        label="Password",
        min_length=8,
        max_length=20,
        required=True
    )

    newsletter = forms.BooleanField(
        label="Subscribe to Newsletter",
        required=False
    )

    skills = forms.MultipleChoiceField(
        label="Programming Skills",
        choices=[
            ("Python", "Python"),
            ("Java", "Java"),
            ("C", "C"),
            ("JavaScript", "JavaScript"),
            ("SQL", "SQL")
        ],
        required=False
    )

    comments = forms.CharField(
        label="Additional Comments",
        max_length=300,
        required=False
    )
